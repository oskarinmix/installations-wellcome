# Caja Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a dual-currency (USD / BCV) cash ledger panel where any user can register deposits, only admins can register withdrawals or change operation status, and a dashboard shows live balances with filterable operation history.

**Architecture:** New Prisma model `CajaOperacion` with two new enums added to the existing schema. Server actions in a new `lib/actions/caja.ts` file follow the same `getUserRole` / `requireAdmin` pattern used throughout the app. The `/caja` page is a client component that fetches balance + operations via those actions and renders balance cards, a filter bar, and an operations table—all consistent with existing page patterns.

**Tech Stack:** Next.js 15 App Router, Prisma (PostgreSQL), TypeScript, Tailwind CSS, shadcn/ui (Card, Dialog, Badge, Button), Lucide icons.

```typescript
export async function getAuthUser(): Promise<{ id: string; role: "admin" | "agent" } | null> {
  try {
    const cookieStore = await cookies();

    const rawToken =
      cookieStore.get("better-auth.session_token")?.value ??
      cookieStore.get("__Secure-better-auth.session_token")?.value;

    if (!rawToken) return null;

    const token = rawToken.split(".")[0];

    const session = await prisma.session.findUnique({
      where: { token },
      select: {
        expiresAt: true,
        user: { select: { id: true, role: true } },
      },
    });

    if (!session || session.expiresAt < new Date()) return null;

    return {
      id: session.user.id,
      role: session.user.role as "admin" | "agent",
    };
  } catch {
    return null;
  }
}
```

```typescript
"use server";

import { prisma } from "@/lib/prisma";
import { getAuthUser, getUserRole } from "@/lib/get-user-role";
import type { CajaOperationType, CajaOperationStatus, Currency } from "@/lib/generated/prisma/client";

export async function getCajaBalance(): Promise<{ usd: number; bcv: number }> {
  const ops = await prisma.cajaOperacion.findMany({
    where: { status: { not: "CANCELADO" } },
    select: { type: true, currency: true, amount: true },
  });

  let usd = 0;
  let bcv = 0;

  for (const op of ops) {
    const sign = op.type === "DEPOSITO" ? 1 : -1;
    if (op.currency === "USD") usd += sign * op.amount;
    else bcv += sign * op.amount;
  }

  return { usd, bcv };
}

export type CajaOperacionRow = {
  id: number;
  type: CajaOperationType;
  status: CajaOperationStatus;
  currency: Currency;
  amount: number;
  description: string;
  createdAt: string;
  createdByName: string;
  updatedByName: string | null;
};

export type CajaFilters = {
  search?: string;
  currency?: "USD" | "BCV";
  type?: "DEPOSITO" | "RETIRO";
  status?: "CARGADO" | "REGISTRADO" | "CANCELADO";
  startDate?: string;
  endDate?: string;
};

export async function getCajaOperaciones(filters: CajaFilters): Promise<CajaOperacionRow[]> {
  const rows = await prisma.cajaOperacion.findMany({
    where: {
      ...(filters.currency ? { currency: filters.currency as Currency } : {}),
      ...(filters.type ? { type: filters.type as CajaOperationType } : {}),
      ...(filters.status ? { status: filters.status as CajaOperationStatus } : {}),
      ...(filters.search
        ? { description: { contains: filters.search, mode: "insensitive" } }
        : {}),
      ...(filters.startDate || filters.endDate
        ? {
            createdAt: {
              ...(filters.startDate ? { gte: new Date(filters.startDate) } : {}),
              ...(filters.endDate ? { lte: new Date(filters.endDate + "T23:59:59.999Z") } : {}),
            },
          }
        : {}),
    },
    include: {
      createdBy: { select: { name: true } },
      updatedBy: { select: { name: true } },
    },
    orderBy: { createdAt: "desc" },
  });

  return rows.map((r) => ({
    id: r.id,
    type: r.type,
    status: r.status,
    currency: r.currency,
    amount: r.amount,
    description: r.description,
    createdAt: r.createdAt.toISOString(),
    createdByName: r.createdBy.name,
    updatedByName: r.updatedBy?.name ?? null,
  }));
}

export type CreateCajaOperacionInput = {
  type: "DEPOSITO" | "RETIRO";
  currency: "USD" | "BCV";
  amount: number;
  description: string;
};

export async function createCajaOperacion(
  data: CreateCajaOperacionInput
): Promise<{ success: boolean; error?: string }> {
  const user = await getAuthUser();
  if (!user) return { success: false, error: "No autenticado" };
  if (data.type === "RETIRO" && user.role !== "admin") {
    return { success: false, error: "Solo el admin puede registrar retiros" };
  }
  if (!data.description.trim()) return { success: false, error: "Descripción requerida" };
  if (data.amount <= 0) return { success: false, error: "Monto debe ser mayor a 0" };

  await prisma.cajaOperacion.create({
    data: {
      type: data.type as CajaOperationType,
      currency: data.currency as Currency,
      amount: data.amount,
      description: data.description.trim(),
      createdById: user.id,
    },
  });

  return { success: true };
}

export async function updateCajaStatus(
  id: number,
  status: "REGISTRADO" | "CANCELADO"
): Promise<{ success: boolean; error?: string }> {
  const user = await getAuthUser();
  if (!user) return { success: false, error: "No autenticado" };
  if (user.role !== "admin") return { success: false, error: "Solo el admin puede cambiar el estado" };

  const op = await prisma.cajaOperacion.findUnique({ where: { id } });
  if (!op) return { success: false, error: "Operación no encontrada" };
  if (op.status === "CANCELADO") return { success: false, error: "No se puede modificar una operación cancelada" };

  await prisma.cajaOperacion.update({
    where: { id },
    data: {
      status: status as CajaOperationStatus,
      updatedById: user.id,
    },
  });

  return { success: true };
}
```

```typescript
"use client";

import { useState } from "react";
import {
  Dialog,
  DialogContent,
  DialogHeader,
  DialogTitle,
  DialogTrigger,
} from "@/components/ui/dialog";
import { Button } from "@/components/ui/button";
import { Label } from "@/components/ui/label";
import { createCajaOperacion } from "@/lib/actions/caja";
import { Plus } from "lucide-react";

interface NuevaOperacionDialogProps {
  isAdmin: boolean;
  onSuccess: () => void;
}

export function NuevaOperacionDialog({ isAdmin, onSuccess }: NuevaOperacionDialogProps) {
  const [open, setOpen] = useState(false);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const [type, setType] = useState<"DEPOSITO" | "RETIRO">("DEPOSITO");
  const [currency, setCurrency] = useState<"USD" | "BCV">("USD");
  const [amount, setAmount] = useState("");
  const [description, setDescription] = useState("");

  async function handleSubmit(e: React.FormEvent) {
    e.preventDefault();
    setError(null);
    const parsed = parseFloat(amount);
    if (isNaN(parsed) || parsed <= 0) {
      setError("Monto inválido");
      return;
    }
    setLoading(true);
    const result = await createCajaOperacion({ type, currency, amount: parsed, description });
    setLoading(false);
    if (!result.success) {
      setError(result.error ?? "Error desconocido");
      return;
    }
    setOpen(false);
    setAmount("");
    setDescription("");
    setType("DEPOSITO");
    setCurrency("USD");
    onSuccess();
  }

  return (
    <Dialog open={open} onOpenChange={setOpen}>
      <DialogTrigger asChild>
        <Button size="sm" className="gap-2">
          <Plus className="h-4 w-4" />
          Nueva Operación
        </Button>
      </DialogTrigger>
      <DialogContent className="sm:max-w-md">
        <DialogHeader>
          <DialogTitle>Nueva Operación de Caja</DialogTitle>
        </DialogHeader>
        <form onSubmit={handleSubmit} className="space-y-4 pt-2">
          <div className="space-y-1.5">
            <Label>Tipo</Label>
            <div className="flex gap-2">
              <button
                type="button"
                onClick={() => setType("DEPOSITO")}
                className={`flex-1 rounded-md border px-3 py-2 text-sm font-medium transition-colors ${
                  type === "DEPOSITO"
                    ? "bg-green-100 border-green-500 text-green-800 dark:bg-green-950 dark:text-green-300"
                    : "hover:bg-accent"
                }`}
              >
                Depósito
              </button>
              <button
                type="button"
                onClick={() => isAdmin && setType("RETIRO")}
                disabled={!isAdmin}
                className={`flex-1 rounded-md border px-3 py-2 text-sm font-medium transition-colors disabled:opacity-40 disabled:cursor-not-allowed ${
                  type === "RETIRO"
                    ? "bg-red-100 border-red-500 text-red-800 dark:bg-red-950 dark:text-red-300"
                    : "hover:bg-accent"
                }`}
              >
                Retiro
              </button>
            </div>
            {!isAdmin && (
              <p className="text-xs text-muted-foreground">Solo admin puede registrar retiros</p>
            )}
          </div>

          <div className="space-y-1.5">
            <Label>Moneda</Label>
            <div className="flex gap-2">
              {(["USD", "BCV"] as const).map((c) => (
                <button
                  key={c}
                  type="button"
                  onClick={() => setCurrency(c)}
                  className={`flex-1 rounded-md border px-3 py-2 text-sm font-medium transition-colors ${
                    currency === c
                      ? "bg-blue-100 border-blue-500 text-blue-800 dark:bg-blue-950 dark:text-blue-300"
                      : "hover:bg-accent"
                  }`}
                >
                  {c === "USD" ? "USD ($)" : "BCV (Bs.)"}
                </button>
              ))}
            </div>
          </div>

          <div className="space-y-1.5">
            <Label htmlFor="amount">Monto</Label>
            <input
              id="amount"
              type="number"
              min="0.01"
              step="0.01"
              value={amount}
              onChange={(e) => setAmount(e.target.value)}
              placeholder="0.00"
              required
              className="w-full h-9 rounded-md border border-input bg-background px-3 text-sm"
            />
          </div>

          <div className="space-y-1.5">
            <Label htmlFor="description">Descripción</Label>
            <input
              id="description"
              type="text"
              value={description}
              onChange={(e) => setDescription(e.target.value)}
              placeholder="Descripción de la operación"
              required
              className="w-full h-9 rounded-md border border-input bg-background px-3 text-sm"
            />
          </div>

          {error && (
            <p className="text-sm text-red-500">{error}</p>
          )}

          <div className="flex gap-2 justify-end pt-2">
            <Button type="button" variant="outline" onClick={() => setOpen(false)}>
              Cancelar
            </Button>
            <Button type="submit" disabled={loading}>
              {loading ? "Guardando..." : "Registrar"}
            </Button>
          </div>
        </form>
      </DialogContent>
    </Dialog>
  );
}
```

```typescript
                  <td className={cn("p-3 text-right font-medium", row.type === "DEPOSITO" ? "text-green-600" : "text-red-500")}>
                    {row.type === "RETIRO" ? "-" : "+"}
                    {row.currency === "USD" ? "$" : "Bs."}{row.amount.toFixed(2)}
                  </td>
                  <td className="p-3 text-muted-foreground">{row.currency}</td>
                  <td className="p-3">
                    <span className={cn("inline-flex items-center rounded-full px-2 py-0.5 text-xs font-medium", STATUS_STYLES[row.status])}>
                      {STATUS_LABELS[row.status]}
                    </span>
                  </td>
                  <td className="p-3 text-muted-foreground">{row.createdByName}</td>
                  {isAdmin && (
                    <td className="p-3">
                      <div className="flex gap-1">
                        {row.status === "CARGADO" && (
                          <button
                            onClick={() => handleStatusChange(row.id, "REGISTRADO")}
                            disabled={actionLoading === row.id}
                            className="rounded px-2 py-1 text-xs font-medium bg-green-100 text-green-800 hover:bg-green-200 dark:bg-green-950 dark:text-green-300 disabled:opacity-50"
                          >
                            {actionLoading === row.id ? "..." : "Registrar"}
                          </button>
                        )}
                        {row.status !== "CANCELADO" && (
                          <button
                            onClick={() => handleStatusChange(row.id, "CANCELADO")}
                            disabled={actionLoading === row.id}
                            className="rounded px-2 py-1 text-xs font-medium bg-red-100 text-red-600 hover:bg-red-200 dark:bg-red-950 dark:text-red-400 disabled:opacity-50"
                          >
                            {actionLoading === row.id ? "..." : "Cancelar"}
                          </button>
                        )}
                      </div>
                    </td>
                  )}
                </tr>
              ))}
            </tbody>
          </table>
        )}
      </div>
    </div>
  );
}
```

## Task 7: Caja Page

**Files:**
- Create: `app/(app)/caja/page.tsx`

- [ ] **Step 1: Create the page**

Create `app/(app)/caja/page.tsx`:

```typescript
"use client";

import { useState, useEffect, useCallback } from "react";
import { BalanceCards } from "@/components/caja/balance-cards";
import { OperacionesTable } from "@/components/caja/operaciones-table";
import { NuevaOperacionDialog } from "@/components/caja/nueva-operacion-dialog";
import { getCajaBalance, getCajaOperaciones } from "@/lib/actions/caja";
import type { CajaOperacionRow, CajaFilters } from "@/lib/actions/caja";
import { getCurrentUserRole } from "@/lib/actions";

export default function CajaPage() {
  const [isAdmin, setIsAdmin] = useState(false);
  const [balance, setBalance] = useState({ usd: 0, bcv: 0 });
  const [rows, setRows] = useState<CajaOperacionRow[]>([]);
  const [filters, setFilters] = useState<CajaFilters>({});

  useEffect(() => {
    getCurrentUserRole().then((res) => {
      if (res?.role === "admin") setIsAdmin(true);
    });
  }, []);

  const loadBalance = useCallback(async () => {
    const b = await getCajaBalance();
    setBalance(b);
  }, []);

  const loadRows = useCallback(async () => {
    const data = await getCajaOperaciones(filters);
    setRows(data);
  }, [filters]);

  useEffect(() => {
    loadBalance();
  }, [loadBalance]);

  useEffect(() => {
    loadRows();
  }, [loadRows]);

  function handleCardSelect(currency: "USD" | "BCV" | null) {
    setFilters((f) => ({ ...f, currency: currency ?? undefined }));
  }

  function handleRefresh() {
    loadBalance();
    loadRows();
  }

  return (
    <div className="space-y-6">
      <div className="flex items-center justify-between">
        <h1 className="text-3xl font-bold flex items-center gap-3">
          <span>🏦</span> Caja
        </h1>
        <NuevaOperacionDialog isAdmin={isAdmin} onSuccess={handleRefresh} />
      </div>

      <BalanceCards
        usd={balance.usd}
        bcv={balance.bcv}
        activeCurrency={filters.currency ?? null}
        onSelect={handleCardSelect}
      />

      <OperacionesTable
        rows={rows}
        filters={filters}
        isAdmin={isAdmin}
        onFiltersChange={setFilters}
        onRefresh={handleRefresh}
      />
    </div>
  );
}
```
